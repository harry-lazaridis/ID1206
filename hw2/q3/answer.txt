Sidstorlek 4KB = 4096 bytes = 2^12 => offset har 12 bitar och sidnumret 32 - 12 = 20bitar
d = 12 bitar
pn = 20 bitar

1) Adress 650003
p = 650003 / 4096 = 158 (158 * 4096 = 647168)
d = 650003 - 657168 = 2835

2) Adress 216202

p = 216202 / 4096 = 52 (52 * 4096 = 212992)
d = 216202 − 212992 = 3210

3) TLB med en nivå

EAT = h * t_TLB + (1 − h)(t_TLB + t_mem)
    = 0.99 * 10 + 0.01 * (10 + 200)
    = 9.9 + 2.1
    = 12 ns

4) TLB med två nivår

EAT = h1 * t1 + (1 − h1) * h2 * (t1 + t2) + (1 − h1)(1 − h2)(t1 + t2 + t_mem)
    = 0.99 * 10 + 0.01 * 0.95 * (10 + 30) + 0.01 * 0.05 * (10 + 30 + 200)
    = 9.9 + 0.38 + 0.12
    = 10.4 ns
